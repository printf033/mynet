function(gen_sample name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE mynet::mynet mynet::mynet_sanitized)
endfunction()

gen_sample(client_tcp)
gen_sample(client_ssl)
gen_sample(client_udp)
gen_sample(reactor_tcp)
gen_sample(reactor_ssl)
gen_sample(reactor_udp)
gen_sample(multiplex_tcp)
gen_sample(proactor_tcp)

gen_sample(proactor_udp)
gen_sample(proactor_multiplex)
